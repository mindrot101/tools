# ─── Admin bootstrap (set these before first boot) ─────────────────────────
# Creates a proper admin on first boot so you're never relying on admin/admin.
# After logging in, delete the default `admin` account from the UI.

VXV_ADMIN_PASS=change-this-now

# ─── Networking ────────────────────────────────────────────────────────────
# Host port to expose the UI on (container always listens on 8080 internally).

# ─── Executor mode ─────────────────────────────────────────────────────
# simulated = canned demo data, no device access (default, safe for demos)
# ssh       = live `show` commands over SSH
# rest      = live AOS-CX REST GET
VXV_EXECUTOR=simulated
